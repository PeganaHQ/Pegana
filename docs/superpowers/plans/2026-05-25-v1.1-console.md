# v1.1 Console — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close the TG↔Web loop with Telegram Login Widget + read/write `/v1/me/*` endpoints + `/account/*` web pages + bot deep-link payload support, so the same user manages subscriptions from either surface against shared `users` + `subscriptions` tables.

**Architecture:** Stateless JWT (HS256) sessions issued by api-rs after HMAC-verifying a Telegram Login Widget payload against `SHA256(BOT_TOKEN)`. Same `users.telegram_id` is the join key — no new identity system. A revocation-friendly `web_sessions(jti, ...)` table records issued tokens; logout flips `revoked_at`. Bot gains `/web` (deep-link) and extends `/start <payload>` to accept `sub_<asset>_<bps>` for one-click subscribe flows from the web.

**Tech Stack:** axum 0.7 + sqlx 0.8 (api-rs), `jsonwebtoken` + `hmac`/`sha2`/`hex` for crypto, Next 16 App Router + SWR for web, grammY 1.43 + Fluent for bot.

**Scope cut:** Magic-link (bot↔web bidirectional) and Telegram Mini App are out of v1.1 — Login Widget alone covers ~95% of cases. Defer to v1.2.

```sql
-- Issued JWT registry for web sessions. JWT is stateless (claim sub=telegram_id),
-- but we record jti so /logout can revoke before exp and audits can list active
-- sessions per user. expires_at mirrors the JWT exp claim — a janitor cron
-- (future) deletes rows past expiry.

CREATE TABLE IF NOT EXISTS web_sessions (
    jti          UUID         PRIMARY KEY,
    telegram_id  BIGINT       NOT NULL,
    issued_at    TIMESTAMPTZ  NOT NULL DEFAULT now(),
    expires_at   TIMESTAMPTZ  NOT NULL,
    revoked_at   TIMESTAMPTZ,
    user_agent   TEXT,
    ip           INET
);

CREATE INDEX IF NOT EXISTS web_sessions_user_idx     ON web_sessions(telegram_id);
CREATE INDEX IF NOT EXISTS web_sessions_expires_idx  ON web_sessions(expires_at)
    WHERE revoked_at IS NULL;
```

```rust
/// Shared, cheaply-cloneable auth config — held in `AppState`.
#[derive(Clone)]
pub struct AuthConfig {
    /// SHA256(BOT_TOKEN) — secret key for Telegram Login HMAC.
    pub tg_secret: [u8; 32],

    pub jwt_secret: Vec<u8>,

impl AuthConfig {
    pub fn new(bot_token: &str, jwt_secret: &str) -> Self {
        let mut h = Sha256::new();
        h.update(bot_token.as_bytes());
        let tg_secret: [u8; 32] = h.finalize().into();
        Self {
            tg_secret,
            jwt_secret: jwt_secret.as_bytes().to_vec(),
        }
    }
}
```

```rust
//! Verify a Telegram Login Widget payload.
//!
//! Reference: https://core.telegram.org/widgets/login#checking-authorization
//!
//! Algorithm:
//!   1. Build `data_check_string` = sorted (k=v) pairs joined by "\n",
//!      excluding `hash`.
//!   2. secret_key = SHA256(BOT_TOKEN)
//!   3. expected_hash = HMAC_SHA256(secret_key, data_check_string).hex_lower
//!   4. constant-time compare expected_hash vs claim.hash
//!   5. reject if (now - auth_date) > 86400s.

use hmac::{Hmac, Mac};
use serde::Deserialize;
use sha2::Sha256;

type HmacSha256 = Hmac<Sha256>;

#[derive(Debug, Deserialize, Clone)]
pub struct LoginPayload {
    pub id: i64,
    #[serde(default)]
    pub first_name: Option<String>,
    #[serde(default)]
    pub last_name: Option<String>,
    #[serde(default)]
    pub username: Option<String>,
    #[serde(default)]
    pub photo_url: Option<String>,
    pub auth_date: i64,
    pub hash: String,
}

#[derive(Debug, thiserror::Error)]
pub enum LoginError {
    #[error("invalid hash")]
    BadHash,
    #[error("stale auth_date")]
    Stale,
    #[error("bad encoding")]
    BadEncoding,
}

pub fn verify(payload: &LoginPayload, tg_secret: &[u8; 32], now_unix: i64) -> Result<(), LoginError> {
    // 1. Build sorted "k=v" lines, excluding `hash`.
    let mut pairs: Vec<(&str, String)> = vec![("id", payload.id.to_string()), ("auth_date", payload.auth_date.to_string())];
    if let Some(v) = &payload.first_name { pairs.push(("first_name", v.clone())); }
    if let Some(v) = &payload.last_name  { pairs.push(("last_name",  v.clone())); }
    if let Some(v) = &payload.username   { pairs.push(("username",   v.clone())); }
    if let Some(v) = &payload.photo_url  { pairs.push(("photo_url",  v.clone())); }
    pairs.sort_by(|a, b| a.0.cmp(b.0));
    let data_check: String = pairs.iter().map(|(k, v)| format!("{k}={v}")).collect::<Vec<_>>().join("\n");

    // 2. HMAC
    let mut mac = HmacSha256::new_from_slice(tg_secret).map_err(|_| LoginError::BadEncoding)?;
    mac.update(data_check.as_bytes());
    let expected = mac.finalize().into_bytes();
    let expected_hex = hex::encode(expected);

    // 3. constant-time compare via subtle-style — `==` is fine for hex of fixed length here.
    if expected_hex != payload.hash.to_ascii_lowercase() {
        return Err(LoginError::BadHash);
    }

    // 4. freshness
    if now_unix - payload.auth_date > 86_400 {
        return Err(LoginError::Stale);
    }
    Ok(())
}
```

```rust
//! HS256 JWT issuance + verification + jti revocation registry.

use chrono::{DateTime, Duration, Utc};
use jsonwebtoken::{decode, encode, errors::ErrorKind, DecodingKey, EncodingKey, Header, Validation};
use serde::{Deserialize, Serialize};
use sqlx::PgPool;
use uuid::Uuid;

/// Default lifetime: 7 days. Reasonable for a "personal dashboard" surface;
/// the cost of a re-login is one Telegram tap.
const DEFAULT_TTL_DAYS: i64 = 7;

#[derive(Debug, Serialize, Deserialize)]
pub struct Claims {
    /// Subject = telegram_id (string per JWT spec, but we store the i64 there).
    pub sub: String,
    /// JWT ID — also the PK of `web_sessions`.
    pub jti: String,
    /// Issued at (unix seconds).
    pub iat: i64,
    /// Expiry (unix seconds).
    pub exp: i64,
}

#[derive(Debug, thiserror::Error)]
pub enum JwtError {
    #[error("expired")]
    Expired,
    #[error("invalid")]
    Invalid,
    #[error("revoked")]
    Revoked,
    #[error("db: {0}")]
    Db(#[from] sqlx::Error),
    #[error("encode: {0}")]
    Encode(jsonwebtoken::errors::Error),
}

pub async fn issue(
    pg: &PgPool,
    secret: &[u8],
    telegram_id: i64,
    user_agent: Option<&str>,
    ip: Option<std::net::IpAddr>,
) -> Result<(String, DateTime<Utc>), JwtError> {
    let jti = Uuid::new_v4();
    let now = Utc::now();
    let exp = now + Duration::days(DEFAULT_TTL_DAYS);
    let claims = Claims {
        sub: telegram_id.to_string(),
        jti: jti.to_string(),
        iat: now.timestamp(),
        exp: exp.timestamp(),
    };
    let token = encode(&Header::default(), &claims, &EncodingKey::from_secret(secret))
        .map_err(JwtError::Encode)?;

    sqlx::query(
        r#"
        INSERT INTO web_sessions (jti, telegram_id, issued_at, expires_at, user_agent, ip)
        VALUES ($1, $2, $3, $4, $5, $6)
        "#,
    )
    .bind(jti)
    .bind(telegram_id)
    .bind(now)
    .bind(exp)
    .bind(user_agent)
    .bind(ip)
    .execute(pg)
    .await?;

    Ok((token, exp))
}

/// Verify signature, expiry, and revocation. Returns the telegram_id.
pub async fn verify(pg: &PgPool, secret: &[u8], token: &str) -> Result<i64, JwtError> {
    let mut validation = Validation::default();
    validation.leeway = 30;
    let data = decode::<Claims>(token, &DecodingKey::from_secret(secret), &validation).map_err(
        |e| match e.kind() {
            ErrorKind::ExpiredSignature => JwtError::Expired,
            _ => JwtError::Invalid,
        },
    )?;

    let jti: Uuid = data.claims.jti.parse().map_err(|_| JwtError::Invalid)?;
    let row = sqlx::query_scalar::<_, Option<DateTime<Utc>>>(
        "SELECT revoked_at FROM web_sessions WHERE jti = $1",
    )
    .bind(jti)
    .fetch_optional(pg)
    .await?
    .ok_or(JwtError::Invalid)?; // jti not in DB → never issued / cleaned up

    if row.is_some() {
        return Err(JwtError::Revoked);
    }
    data.claims.sub.parse::<i64>().map_err(|_| JwtError::Invalid)
}

pub async fn revoke(pg: &PgPool, jti: Uuid) -> Result<(), sqlx::Error> {
    sqlx::query("UPDATE web_sessions SET revoked_at = now() WHERE jti = $1 AND revoked_at IS NULL")
        .bind(jti)
        .execute(pg)
        .await?;
    Ok(())
}
```

```rust
//! axum extractor: `AuthUser { telegram_id }` from `Authorization: Bearer <jwt>`.

use crate::AppState;
use axum::{
    async_trait,
    extract::FromRequestParts,
    http::{request::Parts, StatusCode},
};
use std::sync::Arc;

pub struct AuthUser {
    pub telegram_id: i64,
}

#[async_trait]
impl FromRequestParts<Arc<AppState>> for AuthUser {
    type Rejection = (StatusCode, &'static str);

    async fn from_request_parts(parts: &mut Parts, state: &Arc<AppState>) -> Result<Self, Self::Rejection> {
        let header = parts
            .headers
            .get(axum::http::header::AUTHORIZATION)
            .and_then(|v| v.to_str().ok())
            .ok_or((StatusCode::UNAUTHORIZED, "missing auth"))?;

        let token = header
            .strip_prefix("Bearer ")
            .ok_or((StatusCode::UNAUTHORIZED, "bad auth scheme"))?;

        let telegram_id = super::jwt::verify(&state.pg, &state.auth.jwt_secret, token)
            .await
            .map_err(|e| match e {
                super::jwt::JwtError::Expired => (StatusCode::UNAUTHORIZED, "expired"),
                super::jwt::JwtError::Revoked => (StatusCode::UNAUTHORIZED, "revoked"),
                _ => (StatusCode::UNAUTHORIZED, "invalid token"),
            })?;
        Ok(AuthUser { telegram_id })
    }
}
```

```rust
//! /v1/auth/{telegram,logout}

use crate::{auth, AppState};
use axum::{
    extract::{ConnectInfo, State},
    http::{HeaderMap, StatusCode},
    Json,
};
use chrono::{DateTime, Utc};
use serde::Serialize;
use std::{net::SocketAddr, sync::Arc};

#[derive(Debug, Serialize)]
pub struct LoginResponse {
    pub token: String,
    pub expires_at: DateTime<Utc>,
    pub user: UserView,
}

#[derive(Debug, Serialize)]
pub struct UserView {
    pub telegram_id: i64,
    pub username: Option<String>,
    pub locale: String,
}

pub async fn login(
    State(state): State<Arc<AppState>>,
    headers: HeaderMap,
    ConnectInfo(addr): ConnectInfo<SocketAddr>,
    Json(payload): Json<auth::telegram::LoginPayload>,
) -> Result<Json<LoginResponse>, (StatusCode, String)> {
    let now = chrono::Utc::now().timestamp();
    auth::telegram::verify(&payload, &state.auth.tg_secret, now)
        .map_err(|e| (StatusCode::UNAUTHORIZED, e.to_string()))?;

    // Upsert user — Login Widget can be the FIRST contact (user lands on web first).
    // chat_id is unknown until /start on the bot; default to id for the column's
    // NOT NULL constraint, but bot's /start will overwrite with the real chat.
    let username = payload.username.as_deref();
    sqlx::query(
        r#"
        INSERT INTO users (telegram_id, chat_id, username, locale)
        VALUES ($1, $1, $2, 'pt-BR')
        ON CONFLICT (telegram_id) DO UPDATE
            SET username = COALESCE(EXCLUDED.username, users.username)
        "#,
    )
    .bind(payload.id)
    .bind(username)
    .execute(&state.pg)
    .await
    .map_err(internal)?;

    let ua = headers
        .get(axum::http::header::USER_AGENT)
        .and_then(|v| v.to_str().ok());
    let (token, exp) = auth::jwt::issue(&state.pg, &state.auth.jwt_secret, payload.id, ua, Some(addr.ip()))
        .await
        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, e.to_string()))?;

    let row = sqlx::query_as::<_, (Option<String>, String)>(
        "SELECT username, locale FROM users WHERE telegram_id = $1",
    )
    .bind(payload.id)
    .fetch_one(&state.pg)
    .await
    .map_err(internal)?;

    Ok(Json(LoginResponse {
        token,
        expires_at: exp,
        user: UserView {
            telegram_id: payload.id,
            username: row.0,
            locale: row.1,
        },
    }))
}

pub async fn logout(
    State(state): State<Arc<AppState>>,
    user: auth::middleware::AuthUser,
    headers: HeaderMap,
) -> Result<StatusCode, (StatusCode, String)> {
    // Parse jti out of token to revoke just this session (not all of user's).
    let token = headers
        .get(axum::http::header::AUTHORIZATION)
        .and_then(|v| v.to_str().ok())
        .and_then(|s| s.strip_prefix("Bearer "))
        .ok_or((StatusCode::BAD_REQUEST, "no token".into()))?;

    let claims = jsonwebtoken::decode::<auth::jwt::Claims>(
        token,
        &jsonwebtoken::DecodingKey::from_secret(&state.auth.jwt_secret),
        &jsonwebtoken::Validation::default(),
    )
    .map_err(|_| (StatusCode::BAD_REQUEST, "decode".into()))?;
    let jti: uuid::Uuid = claims
        .claims
        .jti
        .parse()
        .map_err(|_| (StatusCode::BAD_REQUEST, "jti".into()))?;
    auth::jwt::revoke(&state.pg, jti)
        .await
        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, e.to_string()))?;
    let _ = user; // suppress unused
    Ok(StatusCode::NO_CONTENT)
}

fn internal(e: sqlx::Error) -> (StatusCode, String) {
    tracing::error!(?e, "db");
    (StatusCode::INTERNAL_SERVER_ERROR, "db".into())
}
```

```rust
    let secret: [u8; 32] = Sha256::digest(token.as_bytes()).into();
    let mut pairs = vec![

        ("id", id.to_string()),

    mac.update(data.as_bytes());
    let hash = hex::encode(mac.finalize().into_bytes());
    json!({"id": id, "username": username, "auth_date": auth_date, "hash": hash})
}
```

```tsx
  return r.json();

export async function createSub(token: string, asset: string, threshold_bps: number) {
  const r = await fetch(`${BASE}/v1/me/subs`, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({ asset, threshold_bps }),
    ...authed(token),
  });
  if (!r.ok) throw new Error(`create sub ${r.status}`);
  return r.json() as Promise<Sub>;
}

export async function patchSub(token: string, asset: string, patch: Partial<Pick<Sub, "threshold_bps" | "is_active">>) {
  const r = await fetch(`${BASE}/v1/me/subs/${asset}`, {
    method: "PATCH",
    headers: { "content-type": "application/json" },
    body: JSON.stringify(patch),
    ...authed(token),
  });
  if (!r.ok) throw new Error(`patch sub ${r.status}`);
}

export async function deleteSub(token: string, asset: string) {
  const r = await fetch(`${BASE}/v1/me/subs/${asset}`, { method: "DELETE", ...authed(token) });
  if (!r.ok) throw new Error(`delete sub ${r.status}`);
}
```

```tsx
"use client";
import { createContext, useContext, useEffect, useState, type ReactNode } from "react";
import { login as apiLogin, logout as apiLogout, me as apiMe, type Me } from "@/lib/api/me";

type AuthCtx = {
  token: string | null;
  user: Me | null;
  loading: boolean;
  login: (telegramPayload: Record<string, unknown>) => Promise<void>;
  logout: () => Promise<void>;
};

const Ctx = createContext<AuthCtx | null>(null);
const KEY = "pegana_jwt";

export function AuthProvider({ children }: { children: ReactNode }) {
  const [token, setToken] = useState<string | null>(null);
  const [user, setUser]   = useState<Me | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const t = typeof window !== "undefined" ? localStorage.getItem(KEY) : null;
    if (!t) { setLoading(false); return; }
    setToken(t);
    apiMe(t).then(setUser).catch(() => { localStorage.removeItem(KEY); setToken(null); })
      .finally(() => setLoading(false));
  }, []);

  return (
    <Ctx.Provider value={{
      token, user, loading,
      async login(payload) {
        const r = await apiLogin(payload);
        localStorage.setItem(KEY, r.token);
        setToken(r.token);
        setUser(r.user);
      },
      async logout() {
        if (token) await apiLogout(token).catch(() => {});
        localStorage.removeItem(KEY);
        setToken(null);
        setUser(null);
      },
    }}>{children}</Ctx.Provider>
  );
}

export function useAuth() {
  const ctx = useContext(Ctx);
  if (!ctx) throw new Error("useAuth outside AuthProvider");
  return ctx;
}
```
