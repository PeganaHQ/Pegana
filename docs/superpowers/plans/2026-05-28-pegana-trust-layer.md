# Pegana Trust Layer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make every Pegana alert independently reproducible — by humans on Telegram/web and by agents via MCP — so any third party can verify the decision against the methodology that was in force at the time.

**Architecture:** New `pegana-methodology` lib crate holds the pure math. The engine calls into it at runtime; a new `pegana-replay` CLI binary (distributed via cargo-dist to `releases.pegana.xyz`) calls the same functions offline. Each emitted alert writes an append-only `alert_evidence` row to Postgres containing inputs, methodology version, computed outputs, and a sha256 receipt. The evidence URL flows out through dispatcher → Telegram, webhook, MCP responses, and a new public web page at `/audit/[alert_id]`.

**Tech Stack:** Rust 1.90 workspace (existing), `rust_decimal`, `sqlx 0.8`, `tokio 1.40`, `chrono`, `serde_json_canonicalizer 0.3` (RFC 8785), `sha2`, `proptest 1`, `clap 4`, `reqwest 0.12`, `similar 2`. TypeScript with `grammy 1.43` + `@grammyjs/i18n` (bot), `@modelcontextprotocol/sdk` + `zod 3.23` (MCP), Next.js 16 App Router (web). PostgreSQL 16 + TimescaleDB. Binary distribution via `cargo-dist 0.30`. Spec source: `docs/superpowers/specs/2026-05-28-pegana-trust-layer-design.md`.

```rust
//! Threshold resolution per asset class.

use pegana_common::{AssetClass, PegState};
use rust_decimal::Decimal;
use std::collections::HashMap;

/// Yield-bearing wrappers (USDY/sUSD/syrupUSDC/sUSDe/ONyc/pbUSDC) only
/// care about the discount side — market < NAV is the redemption-stress
/// signal. Market > NAV ("premium") is just thin secondary liquidity and
/// shouldn't burn a notification.
pub fn is_direction_sensitive(class: AssetClass) -> bool {
    matches!(class, AssetClass::StableYield)
}

/// Direction-aware variant of `state_for_bps_discount`. For classes where
/// only one side of the spread carries information (currently just
/// `stable_yield`), a premium (negative discount) is normalized to PEGGED.
/// For symmetric classes the function delegates to the abs() form.
pub fn state_for_bps_discount_aware(
    class: AssetClass,
    discount: Decimal,
    thresholds: &HashMap<String, u32>,
) -> PegState {
    if is_direction_sensitive(class) && discount < Decimal::ZERO {
        return PegState::Pegged;
    }
    state_for_bps_discount(discount, thresholds)
}

/// Convert smoothed discount → state using BPS thresholds.
pub fn state_for_bps_discount(discount: Decimal, thresholds: &HashMap<String, u32>) -> PegState {
    let abs_bps = (discount.abs() * Decimal::from(10_000u32)).to_string();
    let abs_bps: u32 = abs_bps
        .split('.')
        .next()
        .and_then(|s| s.parse().ok())
        .unwrap_or(0);

    let drift = thresholds
        .get("drift")
        .or_else(|| thresholds.get("drift_bps"))
        .copied()
        .unwrap_or(20);
    let depeg = thresholds
        .get("depeg")
        .or_else(|| thresholds.get("depeg_bps"))
        .copied()
        .unwrap_or(100);
    let critical = thresholds
        .get("critical")
        .or_else(|| thresholds.get("critical_bps"))
        .copied()
        .unwrap_or(300);

    match abs_bps {
        n if n >= critical => PegState::Critical,
        n if n >= depeg => PegState::Depeg,
        n if n >= drift => PegState::Drift,
        _ => PegState::Pegged,
    }
}

/// CR-based state for hyUSD-style CDP stables.
pub fn state_for_cr(cr: Decimal, thresholds: &HashMap<String, u32>) -> PegState {
    let drift = thresholds
        .get("drift")
        .or_else(|| thresholds.get("cr_drift"))
        .copied()
        .unwrap_or(150);
    let depeg = thresholds
        .get("depeg")
        .or_else(|| thresholds.get("cr_depeg"))
        .copied()
        .unwrap_or(130);
    let critical = thresholds
        .get("critical")
        .or_else(|| thresholds.get("cr_critical"))
        .copied()
        .unwrap_or(110);
    let black_swan = thresholds
        .get("black_swan")
        .or_else(|| thresholds.get("cr_black_swan"))
        .copied()
        .unwrap_or(100);

    let cr_pct = (cr * Decimal::from(100u32))
        .to_string()
        .split('.')
        .next()
        .and_then(|s| s.parse::<i64>().ok())
        .unwrap_or(0)
        .max(0) as u32;

    match cr_pct {
        n if n < black_swan => PegState::BlackSwan,
        n if n < critical => PegState::Critical,
        n if n < depeg => PegState::Depeg,
        n if n < drift => PegState::Drift,
        _ => PegState::Pegged,
    }
}
```

```rust
//! Discount computation and plausibility checks.

use pegana_common::AssetClass;
use rust_decimal::Decimal;

/// Compute the signed discount `1 - market/intrinsic`. For LSTs with a SOL
/// numeraire path available, the SOL-denominated values are preferred
/// (cancels the pyth(SOL/USD) multiplier race).
///
/// Returns `Decimal::ZERO` when intrinsic is zero — the caller is expected
/// to gate on the plausibility check before publishing.
pub fn compute_discount(
    intrinsic: Decimal,
    market: Decimal,
    intrinsic_sol: Option<Decimal>,
    market_sol: Option<Decimal>,
    class: AssetClass,
) -> Decimal {
    if intrinsic.is_zero() {
        return Decimal::ZERO;
    }
    if matches!(class, AssetClass::Lst) {
        if let (Some(i_sol), Some(m_sol)) = (intrinsic_sol, market_sol) {
            if !i_sol.is_zero() {
                return Decimal::ONE - (m_sol / i_sol);
            }
        }
    }
    Decimal::ONE - (market / intrinsic)
}

/// Discount samples with |d| > 1.0 are physically impossible for any
/// tracked class — a market at 2× or 0× intrinsic is upstream noise.
pub fn is_plausible_discount_sample(d: Decimal) -> bool {
```

```rust
//! Hysteresis state-machine step extracted from engine state.rs:258-296.
//! Takes `now` explicitly so the replay CLI is deterministic.

use chrono::{DateTime, Utc};
use pegana_common::PegState;
use serde::{Deserialize, Serialize};

/// Outcome of one hysteresis step. Caller writes these back to its cache.
#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq, Eq)]
pub struct TransitionDecision {
    pub new_last_state: PegState,
    pub new_candidate_state: Option<PegState>,
    pub new_candidate_since: Option<DateTime<Utc>>,
}

/// One hysteresis step. Mirrors engine state.rs::transition exactly,
/// minus the cache mutation (caller is responsible for writing back).
pub fn transition_decide(
    current_last_state: PegState,
    candidate: PegState,
    prev_candidate_state: Option<PegState>,
    prev_candidate_since: Option<DateTime<Utc>>,
    now: DateTime<Utc>,
    confirm_up_secs: i64,
    decay_down_secs: i64,
) -> TransitionDecision {
    let strictness = state_strictness(candidate);
    let current_strictness = state_strictness(current_last_state);

    // Already in the candidate state — clear pending candidate.
    if candidate == current_last_state {
        return TransitionDecision {
            new_last_state: current_last_state,
            new_candidate_state: None,
            new_candidate_since: None,
        };
    }

    // New candidate starts — record it.
    if prev_candidate_state != Some(candidate) {
        return TransitionDecision {
            new_last_state: current_last_state,
            new_candidate_state: Some(candidate),
            new_candidate_since: Some(now),
        };
    }

    // Same candidate persists — check whether it's matured.
    let since = prev_candidate_since.unwrap_or(now);
    let needed = if strictness > current_strictness {
        confirm_up_secs
    } else {
        decay_down_secs
    };
    if (now - since).num_seconds() >= needed {
        TransitionDecision {
            new_last_state: candidate,
            new_candidate_state: None,
            new_candidate_since: None,
        }
    } else {
        TransitionDecision {
            new_last_state: current_last_state,
            new_candidate_state: prev_candidate_state,
            new_candidate_since: prev_candidate_since,
        }
    }
}

fn state_strictness(s: PegState) -> u8 {
    match s {
        PegState::Pegged => 0,
        PegState::Drift => 1,
        PegState::Depeg => 2,
        PegState::Critical => 3,
        PegState::BlackSwan => 4,
        PegState::Unknown => 0,
    }
}
```

```rust
use thiserror::Error;

#[derive(Debug, Error)]
pub enum MethodologyError {
    #[error("failed to parse assets.toml: {0}")]
    AssetsTomlParse(#[from] toml::de::Error),

    #[error("failed to serialize canonical TOML: {0}")]
    AssetsTomlSerialize(#[from] toml::ser::Error),

    #[error("canonical JSON encode failed: {0}")]
    CanonicalJson(#[from] serde_json::Error),
}

/// Parse the assets.toml string, re-serialize in canonical form
/// (sorted keys, normalized whitespace), then sha256.
///
/// Comments and original formatting do not affect the hash; semantic
/// content does. Two TOML files with the same `[[assets]]` entries and
/// thresholds hash to the same value regardless of comment changes.
pub fn canonical_assets_hash(toml_str: &str) -> Result<[u8; 32], MethodologyError> {
    let value: toml::Value = toml::from_str(toml_str)?;
    let canonical = toml::to_string(&value)?;
    let mut hasher = Sha256::new();
    hasher.update(canonical.as_bytes());
    Ok(hasher.finalize().into())
}

/// Canonical sha256 of a receipt's content. Stable across machines and
/// languages — used by the replay CLI to detect tampering.
///
/// Hashes `methodology_version || methodology_git_sha || assets_toml_canonical
/// || inputs_frozen || computed`, all JCS-encoded.
pub fn canonical_receipt_hash(
    methodology_version: &str,
    methodology_git_sha: Option<&str>,
    assets_toml_canonical: &str,
    inputs_frozen: &InputsFrozen,
    computed: &Computed,
) -> Result<[u8; 32], MethodologyError> {
    let mut hasher = Sha256::new();
    hasher.update(methodology_version.as_bytes());
    hasher.update(b"\x00");
    hasher.update(methodology_git_sha.unwrap_or("").as_bytes());
    hasher.update(b"\x00");
    hasher.update(assets_toml_canonical.as_bytes());
    hasher.update(b"\x00");
    let inputs_bytes = serde_json_canonicalizer::to_string(inputs_frozen)?;
    hasher.update(inputs_bytes.as_bytes());
    hasher.update(b"\x00");
    let computed_bytes = serde_json_canonicalizer::to_string(computed)?;
    hasher.update(computed_bytes.as_bytes());
    Ok(hasher.finalize().into())
}

/// Hex-encode a sha256 digest for storage in CHAR(64) columns.

    digest.iter().fold(String::with_capacity(64), |mut acc, b| {
        acc.push_str(&format!("{:02x}", b));
        acc
    })
}
```

```sql
    receipt_sha256        CHAR(64)    NOT NULL,

    created_at            TIMESTAMPTZ NOT NULL DEFAULT NOW()

BEGIN
    RAISE EXCEPTION 'alert_evidence is append-only';

CREATE TRIGGER alert_evidence_no_delete
```

```rust
/// `GET /v1/audit/:alert_id` — human-readable receipt JSON.
#[utoipa::path(
    get,
    path = "/v1/audit/{alert_id}",
    params(("alert_id" = String, Path, description = "Alert UUID")),
    responses(
        (status = 200, description = "Receipt JSON"),
        (status = 404, description = "Alert not found, or evidence missing"),
    ),
    tag = "audit"
)]
pub async fn detail(
    State(state): State<Arc<AppState>>,
    Path(alert_id): Path<String>,
) -> Result<Json<serde_json::Value>, (StatusCode, Json<serde_json::Value>)> {
    let id = Uuid::parse_str(&alert_id).map_err(|_| {
        (
            StatusCode::BAD_REQUEST,
            Json(json!({"error": "invalid_uuid"})),
        )
    })?;

    let row = sqlx::query(
        r#"
        SELECT a.id, a.asset, a.class::text AS class, a.from_state::text AS from_state,
               a.to_state::text AS to_state, a.discount, a.intrinsic_usd, a.market_usd,
               a.confidence, a.detected_at,
               e.methodology_version, e.methodology_git_sha, e.assets_toml_sha256,
               e.inputs_frozen, e.computed, e.replay_artifact, e.receipt_sha256, e.created_at AS evidence_at
        FROM alerts a
        LEFT JOIN alert_evidence e ON e.alert_id = a.id
        WHERE a.id = $1
        "#,
    )
    .bind(id)
    .fetch_optional(&state.pg)
    .await
    .map_err(|e| {
        (
            StatusCode::INTERNAL_SERVER_ERROR,
            Json(json!({"error": "db_error", "message": e.to_string()})),
        )
    })?;

    let row = row.ok_or_else(|| {
        (
            StatusCode::NOT_FOUND,
            Json(json!({"error": "alert_not_found", "alert_id": alert_id})),
        )
    })?;

    let methodology_version: Option<String> = row.try_get("methodology_version").ok();
    if methodology_version.is_none() {
        return Err((
            StatusCode::NOT_FOUND,
            Json(json!({"error": "evidence_missing", "alert_id": alert_id})),
        ));
    }

    Ok(Json(json!({
        "alert": {
            "id": row.get::<Uuid, _>("id").to_string(),
            "asset": row.get::<String, _>("asset"),
            "class": row.get::<String, _>("class"),
            "from_state": row.get::<String, _>("from_state"),
            "to_state": row.get::<String, _>("to_state"),
            "discount": row.get::<sqlx::types::Decimal, _>("discount").to_string(),
            "intrinsic_usd": row.get::<sqlx::types::Decimal, _>("intrinsic_usd").to_string(),
            "market_usd": row.get::<sqlx::types::Decimal, _>("market_usd").to_string(),
            "confidence": row.get::<String, _>("confidence"),
            "detected_at": row.get::<chrono::DateTime<chrono::Utc>, _>("detected_at"),
        },
        "evidence": {
            "methodology_version": methodology_version,
            "methodology_git_sha": row.get::<Option<String>, _>("methodology_git_sha"),
            "assets_toml_sha256": row.get::<String, _>("assets_toml_sha256"),
            "inputs_frozen": row.get::<serde_json::Value, _>("inputs_frozen"),
            "computed": row.get::<serde_json::Value, _>("computed"),
            "has_replay_artifact": !row.get::<Option<serde_json::Value>, _>("replay_artifact").map(|v| v.is_null()).unwrap_or(true),
            "receipt_sha256": row.get::<String, _>("receipt_sha256"),
            "created_at": row.get::<chrono::DateTime<chrono::Utc>, _>("evidence_at"),
        },
    })))
}

/// `GET /v1/audit/:alert_id/replay-bundle` — full self-contained JSON
/// suitable for offline replay via `pegana-replay --bundle ./file.json`.
#[utoipa::path(
    get,
    path = "/v1/audit/{alert_id}/replay-bundle",
    params(("alert_id" = String, Path, description = "Alert UUID")),
    responses(
        (status = 200, description = "Replay bundle JSON"),
        (status = 404, description = "Alert not found, or no artifact (non-critical alert)"),
    ),
    tag = "audit"
)]
pub async fn replay_bundle(
    State(state): State<Arc<AppState>>,
    Path(alert_id): Path<String>,
) -> Result<impl IntoResponse, (StatusCode, Json<serde_json::Value>)> {
    let id = Uuid::parse_str(&alert_id).map_err(|_| {
        (
            StatusCode::BAD_REQUEST,
            Json(json!({"error": "invalid_uuid"})),
        )
    })?;

    let bundle: Option<serde_json::Value> = sqlx::query_scalar(
        "SELECT replay_artifact FROM alert_evidence WHERE alert_id = $1",
    )
    .bind(id)
    .fetch_optional(&state.pg)
    .await
    .map_err(|e| {
        (
            StatusCode::INTERNAL_SERVER_ERROR,
            Json(json!({"error": "db_error", "message": e.to_string()})),
        )
    })?
    .flatten();

    let bundle = bundle.ok_or_else(|| {
        (
            StatusCode::NOT_FOUND,
            Json(json!({"error": "no_replay_artifact", "alert_id": alert_id, "hint": "non-critical alerts (DRIFT) have no artifact"})),
        )
    })?;

    let mut headers = HeaderMap::new();
    headers.insert(
        header::CONTENT_DISPOSITION,
        format!("attachment; filename=\"alert-{}.json\"", alert_id).parse().unwrap(),
    );

    Ok((StatusCode::OK, headers, Json(bundle)))
}
```

```rust
//! pegana-replay — verify a Pegana alert against its receipt.

use anyhow::{anyhow, bail, Context, Result};
use clap::Parser;
use pegana_methodology::{
    canonical_assets_hash, canonical_receipt_hash, methodology_version, Receipt,
};
use std::path::PathBuf;
use uuid::Uuid;

#[derive(Parser, Debug)]
#[command(version, about, long_about = None)]
struct Cli {
    /// Alert UUID to verify by fetching from the API.
    #[arg(long, conflicts_with = "bundle")]
    alert_id: Option<Uuid>,

    /// Path to a local replay bundle JSON.
    #[arg(long, conflicts_with = "alert_id")]
    bundle: Option<PathBuf>,

    /// API base URL.
    #[arg(long, env = "PEGANA_API", default_value = "https://api.pegana.xyz")]
    api_url: String,

    /// Suppress all output except final PASS/FAIL.
    #[arg(long)]
    quiet: bool,
}

#[tokio::main]
async fn main() -> Result<()> {
    let cli = Cli::parse();
    if cli.alert_id.is_none() && cli.bundle.is_none() {
        bail!("provide either --alert-id <UUID> or --bundle <path>");
    }

    let receipt: Receipt = if let Some(id) = cli.alert_id {
        fetch_bundle(&cli.api_url, id).await?
    } else {
        let path = cli.bundle.unwrap();
        let text = std::fs::read_to_string(&path)
            .with_context(|| format!("read bundle {}", path.display()))?;
        serde_json::from_str(&text).context("parse bundle JSON")?
    };

    verify(&receipt, cli.quiet)
}

async fn fetch_bundle(api_url: &str, alert_id: Uuid) -> Result<Receipt> {
    let url = format!("{}/v1/audit/{}/replay-bundle", api_url.trim_end_matches('/'), alert_id);
    let resp = reqwest::get(&url)
        .await
        .with_context(|| format!("GET {}", url))?;
    if !resp.status().is_success() {
        bail!("API returned {}: {}", resp.status(), url);
    }
    let receipt: Receipt = resp.json().await.context("decode bundle JSON")?;
    Ok(receipt)
}

fn verify(receipt: &Receipt, quiet: bool) -> Result<()> {
    // 1) Schema version.
    if receipt.schema_version != "v1" {
        bail!("unsupported schema_version: {}", receipt.schema_version);
    }

    // 2) Methodology version match.
    let cli_version = methodology_version();
    if receipt.methodology_version != cli_version {
        eprintln!(
            "VERSION_MISMATCH: alert uses methodology v{}; this CLI is v{}.",
            receipt.methodology_version, cli_version
        );
        eprintln!(
            "Install the matching version:\n  cargo install pegana-replay-cli --version {}",
            receipt.methodology_version
        );
        std::process::exit(3);
    }

    // 3) Verify assets canonical hash.
    let actual_assets_hash = canonical_assets_hash(&receipt.assets_toml_canonical)
        .map_err(|e| anyhow!("canonical_assets_hash failed: {e}"))?;
    let actual_assets_hex = pegana_methodology::canonical::hex_sha256(actual_assets_hash);
    // Note: assets_toml_canonical embedded in the bundle is already the canonical form;
    // re-hashing it should give the assets_toml_sha256 stored alongside the receipt.

    // 4) Recompute receipt hash.
    let recomputed_hash = canonical_receipt_hash(
        &receipt.methodology_version,
        receipt.methodology_git_sha.as_deref(),
        &receipt.assets_toml_canonical,
        &receipt.inputs_frozen,
        &receipt.expected_computed,
    )
    .map_err(|e| anyhow!("canonical_receipt_hash failed: {e}"))?;
    let recomputed_hex = pegana_methodology::canonical::hex_sha256(recomputed_hash);

    if recomputed_hex != receipt.expected_receipt_sha256 {
        eprintln!("✗ FAIL  receipt sha256 mismatch");
        eprintln!("  expected: {}", receipt.expected_receipt_sha256);
        eprintln!("  computed: {}", recomputed_hex);
        std::process::exit(1);
    }

    if !quiet {
        let inputs = &receipt.inputs_frozen;
        let computed = &receipt.expected_computed;
        println!("✓ PASS  {}  {:?} → {:?}  @ {}",
            inputs.asset, inputs.previous_state, computed.final_state, inputs.now);
        println!(
            "        methodology v{}, assets_toml sha256:{}",
            receipt.methodology_version, &actual_assets_hex[..12],
        );
        println!(
            "        recomputed receipt:{} matches stored receipt:{}",
            &recomputed_hex[..12],
            &receipt.expected_receipt_sha256[..12],
        );
    }
    Ok(())
}
```

```rust
#[utoipa::path(
    get,
    path = "/v1/audit",
    params(
        ("limit" = Option<u32>, Query, description = "Max rows (≤200)"),
        ("exclude_pegged" = Option<bool>, Query, description = "Skip PEGGED→PEGGED rows"),
    ),
    responses((status = 200, description = "Recent audited alerts")),
    tag = "audit"
)]
pub async fn index(
    State(state): State<Arc<AppState>>,
    Query(q): Query<IndexParams>,
) -> Result<Json<Vec<serde_json::Value>>, (StatusCode, Json<serde_json::Value>)> {
    let limit = q.limit.unwrap_or(50).min(200);
    let exclude_pegged = q.exclude_pegged.unwrap_or(true);
    let rows = sqlx::query(
        r#"
        SELECT a.id, a.asset, a.from_state::text AS from_state, a.to_state::text AS to_state,
               a.detected_at, e.methodology_version, e.receipt_sha256
        FROM alerts a
        JOIN alert_evidence e ON e.alert_id = a.id
        WHERE ($1 = false OR a.to_state <> 'PEGGED')
        ORDER BY a.detected_at DESC
        LIMIT $2
        "#,
    )
    .bind(exclude_pegged)
    .bind(limit as i64)
    .fetch_all(&state.pg)
    .await
    .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, Json(serde_json::json!({"error": e.to_string()}))))?;

    let json: Vec<serde_json::Value> = rows.into_iter().map(|row| {
        serde_json::json!({
            "id": row.get::<Uuid, _>("id").to_string(),
            "asset": row.get::<String, _>("asset"),
            "from_state": row.get::<String, _>("from_state"),
            "to_state": row.get::<String, _>("to_state"),
            "detected_at": row.get::<chrono::DateTime<chrono::Utc>, _>("detected_at"),
            "methodology_version": row.get::<String, _>("methodology_version"),
            "receipt_sha256": row.get::<String, _>("receipt_sha256"),
        })
    }).collect();

    Ok(Json(json))
}

#[derive(serde::Deserialize)]
pub struct IndexParams {
    pub limit: Option<u32>,
    pub exclude_pegged: Option<bool>,
}
```

```rust
//! Anchor receipt sha256 on-chain via Solana SPL Memo program.
//! Fire-and-forget — failures are logged + counted, never block alert dispatch.

use anyhow::{Context, Result};
use solana_client::nonblocking::rpc_client::RpcClient;
use solana_sdk::{
    commitment_config::CommitmentConfig, signature::Keypair, signer::Signer,
    transaction::Transaction,
};
use spl_memo::id as memo_program_id;
use sqlx::PgPool;
use std::sync::Arc;
use std::time::Duration;
use tracing::{error, info, warn};
use uuid::Uuid;

const MEMO_FORMAT_VERSION: &str = "pegana-v1";
const COMMIT_TIMEOUT: Duration = Duration::from_secs(60);

pub struct CommitContext {
    pub rpc: Arc<RpcClient>,
    pub keypair: Arc<Keypair>,
    pub pg: PgPool,
}

pub async fn commit_receipt(
    ctx: Arc<CommitContext>,
    alert_id: Uuid,
    methodology_version: String,
    receipt_sha256: String,
) {
    tokio::spawn(async move {
        let result = tokio::time::timeout(COMMIT_TIMEOUT, do_commit(ctx.clone(), alert_id, &methodology_version, &receipt_sha256)).await;
        match result {
            Ok(Ok(sig)) => {
                if let Err(e) = persist_sig(&ctx.pg, alert_id, &sig).await {
                    error!(?e, %alert_id, "failed to persist onchain_tx_sig");
                    metrics::counter!("pegana_onchain_persist_failed_total").increment(1);
                }
                info!(%alert_id, signature = %sig, "receipt committed on-chain");
                metrics::counter!("pegana_onchain_commit_total", "outcome" => "ok").increment(1);
            }
            Ok(Err(e)) => {
                error!(?e, %alert_id, "onchain commit failed");
                metrics::counter!("pegana_onchain_commit_total", "outcome" => "rpc_error").increment(1);
            }
            Err(_) => {
                error!(%alert_id, "onchain commit timeout");
                metrics::counter!("pegana_onchain_commit_total", "outcome" => "timeout").increment(1);
            }
        }
    });
}

async fn do_commit(
    ctx: Arc<CommitContext>,
    alert_id: Uuid,
    methodology_version: &str,
    receipt_sha256: &str,
) -> Result<String> {
    let memo_payload = format!(
        "{}|{}|{}|{}",
        MEMO_FORMAT_VERSION, methodology_version, alert_id, receipt_sha256
    );

    let memo_ix = spl_memo::build_memo(memo_payload.as_bytes(), &[&ctx.keypair.pubkey()]);
    let blockhash = ctx.rpc.get_latest_blockhash().await.context("get_latest_blockhash")?;
    let tx = Transaction::new_signed_with_payer(
        &[memo_ix],
        Some(&ctx.keypair.pubkey()),
        &[ctx.keypair.as_ref()],
        blockhash,
    );

    let sig = ctx
        .rpc
        .send_and_confirm_transaction_with_spinner_and_commitment(
            &tx,
            CommitmentConfig::confirmed(),
        )
        .await
        .context("send_and_confirm")?;

    Ok(sig.to_string())
}

async fn persist_sig(pg: &PgPool, alert_id: Uuid, sig: &str) -> Result<()> {
    sqlx::query(
        "UPDATE alert_evidence SET onchain_tx_sig = $1 WHERE alert_id = $2 AND onchain_tx_sig IS NULL"
    )
    .bind(sig)
    .bind(alert_id)
    .execute(pg)
    .await
    .context("update alert_evidence.onchain_tx_sig")?;
    Ok(())
}

/// Check engine wallet balance at startup. Logs warn if below threshold.
pub async fn check_wallet_balance(rpc: &RpcClient, keypair: &Keypair) -> Result<u64> {
    let lamports = rpc.get_balance(&keypair.pubkey()).await.context("get_balance")?;
    let sol = lamports as f64 / 1_000_000_000.0;
    if sol < 0.5 {
        warn!(pubkey = %keypair.pubkey(), sol, "engine commit wallet balance LOW (<0.5 SOL)");
        metrics::gauge!("pegana_commit_wallet_balance_sol").set(sol);
    } else {
        info!(pubkey = %keypair.pubkey(), sol, "engine commit wallet balance OK");
    }
    metrics::gauge!("pegana_commit_wallet_balance_sol").set(sol);
    Ok(lamports)
}
```

```rust
    Ok(Json(serde_json::json!({
        "tx_sig": sig,
        "explorer_url": format!("https://solscan.io/tx/{}", sig),
    })))
}
```

```typescript
import { Feed } from "feed";

export const revalidate = 60;

const API_BASE = process.env.PEGANA_API ?? "https://api.pegana.xyz";
const SITE = "https://pegana.xyz";

type IndexRow = {
  id: string;
  asset: string;
  from_state: string;
  to_state: string;
  detected_at: string;
  methodology_version: string;
  receipt_sha256: string;
};

export async function GET() {
  const resp = await fetch(`${API_BASE}/v1/audit?limit=50&exclude_pegged=true`, {
    next: { revalidate: 60 },
  });
  if (!resp.ok) {
    return new Response(`API error ${resp.status}`, { status: 502 });
  }
  const audits: IndexRow[] = await resp.json();

  const feed = new Feed({
    title: "Pegana — Recent Audits",
    description: "Verifiable peg-risk alerts for Solana stablecoins, LSTs, and yield-bearing assets.",
    id: `${SITE}/audit`,
    link: `${SITE}/audit`,
    language: "en",
    favicon: `${SITE}/favicon.ico`,
    copyright: `© ${new Date().getFullYear()} Pegana`,
    feedLinks: { atom: `${SITE}/audit.xml` },
    author: { name: "Pegana", link: SITE },
  });

  for (const a of audits) {
    feed.addItem({
      title: `${a.asset}: ${a.from_state} → ${a.to_state}`,
      id: `${SITE}/audit/${a.id}`,
      link: `${SITE}/audit/${a.id}`,
      description: `Methodology v${a.methodology_version}. Receipt sha256: ${a.receipt_sha256.slice(0, 16)}…`,
      date: new Date(a.detected_at),
    });
  }

  return new Response(feed.atom1(), {
    headers: {
      "Content-Type": "application/atom+xml; charset=utf-8",
      "Cache-Control": "public, s-maxage=60, stale-while-revalidate=300",

  });
}
```

```rust
use serde::Serialize;

#[derive(Serialize)]
pub struct AuditCsvRow {
    pub alert_id: String,
    pub asset: String,
    pub class: String,
    pub from_state: String,
    pub to_state: String,
    pub discount: String,
    pub intrinsic_usd: String,
    pub market_usd: String,

    pub methodology_version: String,

    pub receipt_sha256: String,

pub struct CsvParams {
    pub from: DateTime<Utc>,
    pub to: DateTime<Utc>,
    pub asset: Option<String>,
}

pub async fn csv(
    State(state): State<Arc<AppState>>,
    Query(q): Query<CsvParams>,
) -> Result<impl IntoResponse, (StatusCode, Json<serde_json::Value>)> {

    if (q.to - q.from).num_days() > 366 {
        return Err((StatusCode::BAD_REQUEST, Json(serde_json::json!({"error": "date_range_too_large", "max_days": 366}))));
    }

    let pg = state.pg.clone();
    let asset = q.asset.clone();

    let stream = async_stream::stream! {
        let mut cursor: Option<DateTime<Utc>> = Some(q.to);
        const PAGE: i64 = 1000;
        loop {
            let upper = match cursor { Some(c) => c, None => break };
            let rows: Vec<AuditCsvRow> = match asset {
                Some(ref a) => sqlx::query_as!(/* ... */ ).fetch_all(&pg).await,
```

```typescript
import { Feed } from "feed";
import { execSync } from "node:child_process";

export const revalidate = 3600; // 1h
const SITE = "https://pegana.xyz";

type GitCommit = {
  sha: string;
  author: string;
  date: Date;
  message: string;
  diff: string;
};

function gitLogAssetsToml(limit: number): GitCommit[] {
  const out = execSync(
    `git -C ${process.cwd()}/.. log --follow --pretty=format:'%H%x00%an%x00%aI%x00%s' -n ${limit} -p assets.toml`,
    { encoding: "utf-8" }
  );
  const commits: GitCommit[] = [];
  const blocks = out.split(/\n(?=[0-9a-f]{40}\x00)/);
  for (const block of blocks) {
    const [header, ...rest] = block.split("\n");
    const [sha, author, dateStr, message] = header.split("\x00");
    if (!sha) continue;
    const diff = rest.join("\n");
    // Only keep diffs that touched a threshold or an asset entry header.
    if (!/thresholds_(bps|cr)|^\+\[\[assets\]\]|^-\[\[assets\]\]/m.test(diff)) continue;
    commits.push({ sha, author, date: new Date(dateStr), message, diff });
  }
  return commits;
}

export async function GET() {
  const commits = gitLogAssetsToml(100);

  const feed = new Feed({
    title: "Pegana — Calibration Changes",
    description: "Threshold and asset config mutations in assets.toml",
    id: `${SITE}/calibration`,
    link: `${SITE}/methodology#changelog`,
    language: "en",
    feedLinks: { atom: `${SITE}/calibration.xml` },
    author: { name: "Pegana", link: SITE },
  });

  for (const c of commits) {
    feed.addItem({
      title: `Calibration · ${c.message}`,
      id: `${SITE}/calibration#${c.sha}`,
      link: `https://github.com/lrafasouza/pegana/commit/${c.sha}`,
      description: c.diff.slice(0, 1000) + (c.diff.length > 1000 ? "\n...[truncated]" : ""),
      author: [{ name: c.author }],
      date: c.date,
    });
  }

  return new Response(feed.atom1(), {
    headers: {
      "Content-Type": "application/atom+xml; charset=utf-8",
      "Cache-Control": "public, s-maxage=3600, stale-while-revalidate=7200",
    },
  });
}
```
